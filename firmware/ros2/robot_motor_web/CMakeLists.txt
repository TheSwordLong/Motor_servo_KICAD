cmake_minimum_required(VERSION 3.10)
project(robot_motor_web)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/robot_motor_web DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY robot_motor_web/static DESTINATION share/${PROJECT_NAME})
install(FILES requirements.txt DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_pytest REQUIRED)
 ament_add_pytest_test(web_tests test/test_web.py TIMEOUT 240)
 ament_add_pytest_test(web_ros_tests test/test_ros_integration.py TIMEOUT 300)
endif()
ament_package()
